# hosts.txt: handle hosts, i.e. domains that give out handles to unrelated people.

# customDomain = FALSE. Every other valid handle counts as a custom domain (TRUE).
#
# One domain per line. Blank lines and anything after # are ignored.
# "*.bsky.social", ".bsky.social" and "bsky.social" all mean the same thing.

# after each entry is how many sampled accounts used it. There is no official
# directory of handle hosts, so add any new ones you come across.

# ft.com, house.gov, theguardian.com, afp.com ...) are deliberately NOT listed:
# those accounts really do have a custom domain, so they stay TRUE.
